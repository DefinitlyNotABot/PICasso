\section{Einführung}
% ===========================================================

\begin{frame}{Was ist PICasso?}
    \begin{columns}
        \begin{column}{0.6\textwidth}
            \textbf{PICasso} ist ein Simulator für PIC-Midrange-Mikro\-controller, geschrieben in C++17.
            \vspace{0.5em}
            \begin{itemize}
                \item Liest assemblierte PIC-Programme im \texttt{.LST}-Format
                \item Führt Instruktionen Schritt für Schritt aus
                \item Simuliert die vollständige PIC-Hardware:
                    \begin{itemize}
                        \item 2-Bank-RAM mit Registerspiegelung
                        \item 8-Bit-W-Akkumulator
                        \item 8-Level-Hardware-Stack
                        \item Timer / Prescaler-Subsystem
                        \item Alle 35 PIC-Midrange-Instruktionen
                    \end{itemize}
                \item Interaktive ncurses Terminal-UI
            \end{itemize}
        \end{column}
        \begin{column}{0.4\textwidth}
            \centering
            \includegraphics[width=\textwidth]{./images/pic2.png}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Zweck \& Motivation}
    \begin{block}{Problem}
        PIC-Assemblerprogramme ohne physische Hardware testen und debuggen.
    \end{block}
    \vspace{0.5em}
    \begin{block}{Lösung}
        PICasso als Lern- und Entwicklungswerkzeug für eingebettete Systeme.
    \end{block}
    \vspace{0.5em}
    \begin{itemize}
        \item Programme schrittweise oder im Dauerlauf ausführen
        \item RAM-Inhalte inspizieren \emph{und} zur Laufzeit modifizieren
        \item Einzelne Bits umschalten
        \item Simulation läuft in eigenem Thread -- UI bleibt reagierend
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Domain Code -- Beispiel: \texttt{Stack}}
    \textbf{Domain Code} bildet Hardware-Logik ab, unabhängig von Frameworks.
    \vspace{0.5em}
    \begin{columns}
        \begin{column}{0.55\textwidth}
\begin{lstlisting}[language=picassocpp]
class Stack {
  private:
    uint8_t stack[8];
    uint8_t stackPointer;
  public:
    Stack();
    void push(uint8_t value);
    uint8_t pop();
    void reset();
};
\end{lstlisting}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{itemize}
                \item Modelliert 8-Level-Hardware-Stack des PIC
                \item Keine externen Abhängigkeiten
                \item Nur \texttt{push()}, \texttt{pop()}, \texttt{reset()}
                \item Rein fachliche Hardware-Logik
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Dependency Rule -- Positiv-Beispiel}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Abhängigkeit korrekt von außen nach innen:}
            \begin{itemize}
                \item \texttt{Register} (Domain) kennt nur \texttt{Byte}
                \item \texttt{Byte} kennt nur \texttt{Nibble} und \texttt{Bit}
                \item \texttt{Ram} hängt von \texttt{Register} ab --\\
                    \texttt{Register} weiß nichts von \texttt{Ram}
            \end{itemize}
            \vspace{0.5em}
            Alle Klassen innerhalb der Domain-Schicht.\\
            \textbf{Pfeil zeigt immer nach innen} \checkmark
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=\textwidth]{../Docu/UML/2.png}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Dependency Rule -- Negativ-Beispiel}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Problem:} \texttt{Instruction} (Application Layer) hält ein \textbf{statisches} Feld:
            \vspace{0.3em}
            \begin{center}
                \texttt{static shared\_ptr<MemoryInterface>}\\
                \texttt{memoryInterface}
            \end{center}
            \vspace{0.3em}
            \begin{itemize}
                \item Wird von \texttt{PIC} zur Laufzeit gesetzt
                \item Implizite Laufzeit-Abhängigkeit: innen $\leftarrow$ außen
                \item \texttt{Instruction} sollte \texttt{PIC} nicht kennen
            \end{itemize}
            \vspace{0.3em}
            \textbf{Lösung:} Dependency Injection im Konstruktor
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=\textwidth]{../Docu/UML/3.png}
        \end{column}
    \end{columns}
\end{frame}

\section{SOLID-Prinzipien}
% ===========================================================

\begin{frame}{SRP -- Positiv: \texttt{Stack}}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Single Responsibility Principle}:\\
            Genau \emph{ein} Änderungsgrund.
            \vspace{0.6em}
            \begin{block}{Klasse \texttt{Stack}}
                Eine einzige Verantwortlichkeit:\\
                Verwaltung eines 8-Level-LIFO-Stacks
            \end{block}
            \vspace{0.4em}
            \begin{itemize}
                \item Nur \texttt{push()}, \texttt{pop()}, \texttt{reset()}
                \item Kein Rendering, kein Logging, kein Parsing
                \item Änderungsgrund: nur Stack-Logik des PIC ändert sich
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=\textwidth]{../Docu/UML/4.png}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{OCP -- Positiv: \texttt{Instruction}-Hierarchie}
    \textbf{Open/Closed Principle}: Offen für Erweiterung, geschlossen für Modifikation.
    \vspace{0.4em}
    \begin{columns}
        \begin{column}{0.55\textwidth}
\begin{lstlisting}[language=picassocpp]
class Instruction {
  public:
    // offen fuer Erweiterung:
    virtual uint16_t execute() = 0;
    virtual string getName() = 0;
    virtual ~Instruction() = default;
};

// Neue Instruktion ohne Aenderung der Basis:
class ADDWF : public Arithmetic {
  public:
    uint16_t execute() override;
    string getName() override;
};
\end{lstlisting}
        \end{column}
        \begin{column}{0.45\textwidth}

\begin{frame}[fragile]{OCP -- Negativ: \texttt{Compiler} Switch-Kaskade}
    \begin{columns}
        \begin{column}{0.55\textwidth}
\begin{lstlisting}[language=picassocpp, basicstyle=\ttfamily\tiny]
unique_ptr<Instruction>
Compiler::getInstruction(uint16_t instr)
{
    switch(instr) {
      case 0x0064:
        return make_unique<CLRWDT>(instr);
      case 0x0009:
        return make_unique<RETFIE>(instr);
      case 0x0008:
        return make_unique<RETURN>(instr);
      // ~35 weitere Cases...
    }
}
\end{lstlisting}
        \end{column}
        \begin{column}{0.45\textwidth}
            \textbf{Problem:}\\
            Jede neue Instruktion erfordert Änderung an \texttt{getInstruction()} -- Verstoß gegen OCP.
            \vspace{0.5em}

            \textbf{Lösung:} Registry-Pattern
\begin{lstlisting}[language=picassocpp, basicstyle=\ttfamily\tiny]
// Jede Instruktion registriert sich selbst:
static bool registered = []() {
    registry[0x0700] = [](uint16_t op) {
        return make_unique<ADDWF>(op);
    };
    return true;
}();
\end{lstlisting}
        \end{column}
    \end{columns}
    \begin{center}
        \includegraphics[width=0.55\textwidth]{../Docu/UML/7.png}
    \end{center}
\end{frame}

\begin{frame}[fragile]{DIP -- Positiv: \texttt{ALU} hängt von Abstraktion ab}
    \textbf{Dependency Inversion Principle}: Abhängig von Abstraktionen, nicht von Details.
    \vspace{0.5em}
    \begin{columns}
        \begin{column}{0.5\textwidth}
\begin{lstlisting}[language=picassocpp]
// ALU kennt nur die Abstraktion:
class ALU {
  public:
    uint16_t executeInstruction(
        Instruction& instruction)
    {
        return instruction.execute();
    }
};
\end{lstlisting}
            \vspace{0.3em}
            ALU hängt von \texttt{Instruction} (Abstraktion), nicht von \texttt{ADDWF}, \texttt{SUBWF} etc. \checkmark
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Negativ-Beispiel: \texttt{PIC}}\\
            \vspace{0.3em}
            Hängt direkt von konkreter \texttt{Register}-Klasse ab.
            Keine Abstraktionsebene, gegen die programmiert wird.

uint8_t Ram::readRegister(uint8_t addr) {
    if (isOutOfScope(addr))
        throw runtime_error("Out of range");
    if (!memory[bank][addr])
        throw runtime_error("Nullpointer");
    return memory[bank][addr]->readByte();
}
uint8_t Ram::readRegister(uint8_t addr,
                          bool bank) {
    if (isOutOfScope(addr))

uint8_t Ram::readRegister(uint8_t addr) {

\section{Entwurfsmuster}
% ===========================================================

\begin{frame}{Entwurfsmuster 1: Kompositum Pattern}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Memory-Hierarchie als Composite:}
            \vspace{0.5em}
            \begin{itemize}
                \item \texttt{Bit} -- ein boolscher Wert
                \item \texttt{Nibble} -- Array aus 4 \texttt{Bit}-Objekten
                \item \texttt{Byte} -- 2 \texttt{Nibble}-Objekte (High + Low)
                \item \texttt{Register} -- 1 \texttt{Byte}-Objekt
            \end{itemize}
            \vspace{0.5em}
            Jede Ebene bietet Zugriffsmethoden:\\
            \texttt{getBit()}, \texttt{getNibble()}, \texttt{getByte()}
            \vspace{0.3em}

            \textbf{Begründung:} PIC operiert auf Bit-, Nibble- und Byte-Ebene gleichzeitig (Status-Flags, BCD, Registerwerte).
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=0.75\textwidth]{../Docu/UML/18.png}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Entwurfsmuster 2: Strategy Pattern}
    \textbf{\texttt{Instruction}-Hierarchie als Strategy:}
    \vspace{0.3em}
    \begin{columns}
        \begin{column}{0.55\textwidth}
\begin{lstlisting}[language=picassocpp]
// Strategy-Interface:
class Instruction {
  virtual uint16_t execute() = 0;
};

// Konkrete Strategien:
class ADDWF : public Arithmetic {
  uint16_t execute() override;
};
class GOTO : public Jump {
  uint16_t execute() override;
};

// Context (ALU):
class ALU {
  uint16_t executeInstruction(
      Instruction& instr) {
    return instr.execute(); // polymorph
  }
};
\end{lstlisting}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{itemize}
                \item 35+ Instruktionen, alle mit gleichem Lebenszyklus
                \item ALU kennt nur die Abstraktion
                \item Neue Instruktion = neue Klasse, kein bestehender Code ändert sich
            \end{itemize}
            \centering
            \includegraphics[height=3.5cm, angle=90]{../Docu/UML/19.png}
        \end{column}

\section{Fazit}

\begin{frame}{Herausforderungen \& Lessons Learned}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Herausforderungen:}
            \begin{itemize}
                \item Thread-Synchronisation zwischen Simulations- und UI-Thread
                \item Statisches \texttt{MemoryInterface} in \texttt{Instruction} -- technische Schuld
                \item \texttt{PIC}-Klasse als God Object entstanden durch organisches Wachstum
                \item Vollständige Abdeckung aller 35 Instruktionen in Tests
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Lessons Learned:}
            \begin{itemize}
                \item Clean Architecture zahlt sich bei Testbarkeit aus: \texttt{MemoryInterface}-Mock einfach zu erstellen
                \item Strategy Pattern für Instruktions-Dispatch war entscheidend für Erweiterbarkeit
                \item Frühe Refactorings (Extract Class, Replace Conditional) verbessern Wartbarkeit spürbar
                \item Datengetriebene Tests mit JSON reduzieren Testboilerplate erheblich
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
